import React, { useState, useEffect } from 'react';

interface BookingItem {
    id: number;
    room_number: string;
    check_in_date: string;
    status: string;
    total_price: number;
}

const BookingList: React.FC<{ hotelId: string }> = ({ hotelId }) => {
    const [bookings, setBookings] = useState<BookingItem[]>([]);
    const [loading, setLoading] = useState(true);

    useEffect(() => {
        // Fetch all bookings for the specific tenant/hotel ID
        console.log(`Fetching bookings for Hotel ${hotelId}...`);
        // Simulate API call to /api/v1/hotels/${hotelId}/bookings
        setBookings([
            { id: 101, room_number: '101', check_in_date: '2024-11-05', status: 'CONFIRMED', total_price: 350.00 },
            { id: 102, room_number: '205', check_in_date: '2024-11-10', status: 'PENDING', total_price: 180.00 }
        ]);
        setLoading(false);
    }, [hotelId]);

    if (loading) return <p className="text-center py-6">Loading reservation data...</p>;

    return (
        <div className="overflow-x-auto">
            <table className="min-w-full divide-y divide-gray-200">
                <thead className="bg-gray-50">
                    <tr>
                        <th className="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Booking ID</th>
                        <th className="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Room</th>
                        <th className="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Check-in Date</th>
                        <th className="px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">Status</th>
                        <th className="px-6 py-3 text-right text-xs font-medium text-gray-500 uppercase tracking-wider">Total Price</th>
                    </tr>
                </thead>
                <tbody className="bg-white divide-y divide-gray-200">
                    {bookings.map((booking) => (
                        <tr key={booking.id} className="hover:bg-blue-50">
                            <td className="px-6 py-4 text-sm font-medium">{booking.id}</td>
                            <td className="px-6 py-4 text-sm">{booking.room_number}</td>
                            <td className="px-6 py-4 text-sm">{booking.check_in_date}</td>
                            <td className="px-6 py-4">
                                <span className={`px-3 inline-flex items-center rounded-full text-xs font-semibold ${
                                    booking.status === 'CONFIRMED' ? 'bg-green-100 text-green-800' : 'bg-yellow-100 text-yellow-800'
                                }`}>
                                    {booking.status}
                                </span>
                            </td>
                            <td className="px-6 py-4 text-sm text-right font-medium">${booking.total_price.toFixed(2)}</td>
                        </tr>
                    ))}
                </tbody>
            </table>
        </div>
    );
};

export default BookingList;
